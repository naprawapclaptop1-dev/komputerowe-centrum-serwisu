       <p data-template-id="term-post-text" class="canva-text mt-3 body-copy">Wstępna kontrola sprzętu przy uruchamianiu. Wynik może być sygnalizowany kodem, dźwiękiem lub kontrolką.</p>

      <article class="path-card">
       <h3 data-template-id="term-uefi-title" class="canva-text heading">UEFI</h3>
       <p data-template-id="term-uefi-text" class="canva-text mt-3 body-copy">Oprogramowanie uruchamiające komputer przed systemem. Może pokazywać wykryte dyski i ustawienia startu.</p>

document.addEventListener('DOMContentLoaded', () => {
  const search = document.getElementById('fault-search');
  const items = [...document.querySelectorAll('#fault-list .fault')];
  const count = document.getElementById('result-count');
  const empty = document.getElementById('no-results');
  const normalize = value => value.toLocaleLowerCase('pl-PL').normalize('NFD').replace(/[\u0300-\u036f]/g, '').replace(/ł/g, 'l');
  function filter() {
    const query = normalize(search.value.trim());
    let visible = 0;
    items.forEach(item => {
      const matches = !query || normalize(item.dataset.search).includes(query);
      item.hidden = !matches;
      if (matches) visible++;
    });
    count.textContent = visible === 1 ? '1 pasujący temat' : `${visible} pasujących tematów`;
    empty.hidden = visible !== 0;
  }
  search.addEventListener('input', filter);
  document.getElementById('clear-search').addEventListener('click', () => {
    search.value = '';
    filter();
    search.focus();
  });
  filter();
});
