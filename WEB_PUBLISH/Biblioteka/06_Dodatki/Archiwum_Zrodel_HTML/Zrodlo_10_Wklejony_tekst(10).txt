    cards.push(card);

  const normalize = text => text.toLocaleLowerCase('pl-PL').normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '').replace(/ł/g, 'l');
  const search = document.getElementById('lesson-search');
  function filterLessons() {
    const terms = normalize(search.value.trim()).split(/\s+/).filter(Boolean);
    let shown = 0;
    cards.forEach(card => {
      card.hidden = !terms.every(term => normalize(card.dataset.search).includes(term));
      if (!card.hidden) shown++;
    });
    document.getElementById('search-count').textContent = `Widoczne lekcje: ${shown} z ${cards.length}`;
    document.getElementById('no-lessons').hidden = shown !== 0;
  }
  search.addEventListener('input', filterLessons);
  filterLessons();

  const status = document.getElementById('data-status');
  const visitStatus = document.getElementById('visit-status');
  const retry = document.getElementById('retry-button');
  let backendRecords = [];
  let initialized = false;
  let ready = false;
  let loadTimer;
  let attempt = 0;
  const handler = {
    onDataChanged(data) {
      clearTimeout(loadTimer);
      backendRecords = data;
      ready = true;
      document.getElementById('visit-count').textContent =
        String(data.filter(record => record.type === 'visit').length);
      retry.hidden = true;
      if (status.textContent === 'Łączenie z Canva Sheet…') status.textContent = '';
    }
  };

  async function initialize() {
    const currentAttempt = ++attempt;
    clearTimeout(loadTimer);
    initialized = false;
    ready = false;
    retry.hidden = true;
    status.textContent = 'Łączenie z Canva Sheet…';
    loadTimer = setTimeout(() => {
      if (currentAttempt !== attempt || ready) return;
      status.textContent = 'Nie udało się wczytać danych. Spróbuj ponownie.';
      retry.hidden = false;
    }, 8000);
    try {
      const result = await window.dataSdk.init(handler);
      if (currentAttempt !== attempt) return;
      if (!result.isOk) {
        clearTimeout(loadTimer);
        status.textContent = 'Nie udało się połączyć z Canva Sheet. Spróbuj ponownie.';
        retry.hidden = false;
        return;
      }
      initialized = true;
    } catch (_) {
      if (currentAttempt !== attempt) return;
      clearTimeout(loadTimer);
      status.textContent = 'Nie udało się połączyć z Canva Sheet. Spróbuj ponownie.';
      retry.hidden = false;
    }
  }
  retry.addEventListener('click', initialize);

  document.getElementById('visit-button').addEventListener('click', async event => {
    const button = event.currentTarget;
    if (!initialized || !ready) {
      visitStatus.textContent = 'Poczekaj na połączenie z Canva Sheet.';
      return;
    }
    if (backendRecords.length >= 999) {
      visitStatus.textContent = 'Osiągnięto limit 999 zapisów.';
      return;
    }
    button.disabled = true;
    button.setAttribute('aria-busy', 'true');
    visitStatus.textContent = 'Zapisywanie użycia…';
    try {
      const result = await window.dataSdk.create({
        type: 'visit', message: '', email: '', created_at: new Date().toISOString()
      });
      visitStatus.textContent = result.isOk
        ? 'Dziękujemy! Użycie podręcznika zostało zapisane.'
        : 'Nie można potwierdzić zapisu. Sprawdź licznik przed ponowną próbą.';
    } catch (_) {
      visitStatus.textContent = 'Nie można potwierdzić zapisu. Sprawdź licznik przed ponowną próbą.';
    } finally {
      button.disabled = false;
      button.removeAttribute('aria-busy');
    }
  });

    }
    if (!initialized || !ready) {
      status.textContent = 'Poczekaj na połączenie z Canva Sheet lub spróbuj ponownie.';
      return;
    }
    if (backendRecords.length >= 999) {
      status.textContent = 'Osiągnięto limit 999 zapisów.';
      return;
    }
    button.disabled = true;
    button.setAttribute('aria-busy', 'true');
    status.textContent = 'Zapisywanie opinii…';
    try {
      const result = await window.dataSdk.create({
        type: 'feedback', message, email, created_at: new Date().toISOString()
      });
      if (result.isOk) {
        status.textContent = 'Opinia została zapisana w Canva Sheet. Nie została wysłana e-mailem.';
        document.getElementById('feedback-form').reset();
      } else {
        status.textContent = 'Nie można potwierdzić zapisu. Sprawdź dane przed ponowną próbą.';
      }
    } catch (_) {
      status.textContent = 'Nie można potwierdzić zapisu. Sprawdź dane przed ponowną próbą.';
    } finally {
      button.disabled = false;
      button.removeAttribute('aria-busy');
    }
  });

  document.getElementById('mail-button').addEventListener('click', () => {
    const message = document.getElementById('feedback-message').value.trim();
    const link = document.createElement('a');
    link.href = 'mailto:?subject=' +
      encodeURIComponent('Opinia o podręczniku Napraw komputer od zera') +
      '&body=' + encodeURIComponent(message || 'Moja propozycja dotycząca podręcznika:\n');
    link.click();
  });

  initialize();
});
