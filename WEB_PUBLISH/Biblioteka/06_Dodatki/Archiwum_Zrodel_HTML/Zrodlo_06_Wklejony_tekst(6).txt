document.addEventListener('DOMContentLoaded', () => {
  const search = document.getElementById('fault-search');
  const faults = [...document.querySelectorAll('#fault-list .fault')];
  const count = document.getElementById('result-count');
  const empty = document.getElementById('no-results');
  const normalize = value => value.toLocaleLowerCase('pl-PL').normalize('NFD').replace(/[\u0300-\u036f]/g, '').replace(/ł/g, 'l');
  function filter() {
    const words = normalize(search.value.trim()).split(/\s+/).filter(Boolean);
    let visible = 0;
    faults.forEach(item => {
      const match = words.every(word => normalize(item.dataset.search).includes(word));
      item.hidden = !match;
      if (match) visible++;
    });
    count.textContent = visible === 1 ? '1 pasujący temat' : `${visible} pasujących tematów`;
    empty.hidden = visible !== 0;
  }
  search.addEventListener('input', filter);
  document.getElementById('clear-search').addEventListener('click', () => {
    search.value = '';
    filter();
    search.focus();
  });
  filter();

  const canvas = document.getElementById('instruction-canvas');
  const ctx = canvas.getContext('2d');
  const status = document.getElementById('video-status');
  const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
  const topics = {
    power: {title:'Droga zasilania',steps:['Sprawdź gniazdko','Sprawdź przewód','Potwierdź zgodność','Obserwuj reakcję']},

  };

  function draw() {
    const data=topics[topic], active=Math.min(3,Math.floor(frame/40)%4);
    ctx.fillStyle='#102c2b';ctx.fillRect(0,0,960,540);
    ctx.fillStyle='#163d39';ctx.beginPath();ctx.arc(830,58,230,0,Math.PI*2);ctx.fill();
    ctx.fillStyle='#9ed2ba';ctx.font='700 19px "DM Sans",sans-serif';ctx.fillText('NAPRAW KOMPUTER OD ZERA / ANIMACJA',48,65);
    ctx.fillStyle='#ffffff';ctx.font='600 42px "Fraunces",serif';ctx.fillText(data.title,48,134);
    ctx.fillStyle='#b9d7cc';ctx.font='500 17px "DM Sans",sans-serif';ctx.fillText('Sprawdzaj po kolei. Przy zagrożeniu przerwij pracę.',48,170);
    data.steps.forEach((step,i)=>{
      const x=42+i*230;
      box(x,225,210,167,18,i<=active?'#e9eee8':'#244642');
      box(x+17,243,40,40,10,i<=active?'#11695e':'#48645e');
      ctx.fillStyle='#fff';ctx.font='700 20px "DM Sans",sans-serif';ctx.fillText(String(i+1),x+30,270);
      ctx.fillStyle=i<=active?'#173c38':'#d4e5dc';ctx.font='600 19px "DM Sans",sans-serif';
      let line='',lines=[];
      step.split(' ').forEach(word=>{const next=line?line+' '+word:word;if(ctx.measureText(next).width>175&&line){lines.push(line);line=word;}else line=next;});
      lines.push(line);lines.forEach((part,j)=>ctx.fillText(part,x+17,323+j*27));
      if(i<3){ctx.fillStyle='#d99358';ctx.beginPath();ctx.moveTo(x+212,302);ctx.lineTo(x+228,312);ctx.lineTo(x+212,322);ctx.fill();}

  }
  function pause(){playing=false;cancelAnimationFrame(animationId);}
  function tick(time){
    if(!playing)return;
    if(time-last>=40){frame++;last=time;draw();}
    animationId=requestAnimationFrame(tick);
  }
  document.querySelectorAll('.topic-button').forEach(button=>button.addEventListener('click',()=>{
